# Locomotion-only subset for PMG threshold calibration.
# Format: <filename relative to BVH dir> <action label>
# Lines starting with '#' and blank lines are ignored.
# Action label groups same-action pairs; edit freely.
#
# Usage:
#   pmg_cli --calibrate-thresholds BVH locomotion_manifest.txt
#
# Rationale: PMG transitions are meaningful only between compatible locomotion
# cycles. The mixed action clips (kick / slap / vault / sneeze ...) are not
# loopable locomotion and are excluded here.

# --- explicit loop clips (lowest loop_gap; true cycles) ---
SneakLoopA.bvh   sneak
StrutLoopA.bvh   strut
WalkLoopA.bvh    walk

# --- steady walks (treat as walk gait) ---
smoothWalk.bvh               walk
walkReallySmooth.bvh         walk
walkStraightTwiceAsFast.bvh  walk

# --- curved / spiral walks (still walk gait, different trajectory) ---
walkCurve.bvh        walk
walkCurvedLong.bvh   walk
walkMoreCurve.bvh    walk
walkTightCurve.bvh   walk
walkSpiral.bvh       walk
walkFastSpiral.bvh   walk

# --- jog + transition ---
jogCurve.bvh    jog
walkToJog.bvh   walk

# --- standing (near-static reference) ---
standStill.bvh         stand
standReallyStill.bvh   stand
